    /// 負の数量を持つ項目が含まれる場合。入力検査はこのメソッドが担当する。

        throw new NotImplementedException();
    }
}

internal sealed class ReportSink
{
    public void Write(string text) => Console.WriteLine(text);
}

internal sealed class ReportService
{
    private readonly ItemStore _store;
    private readonly ReportFormatter _formatter;
    private readonly ReportSink _sink;

    public ReportService(ItemStore store, ReportFormatter formatter, ReportSink sink)
    {
        _store = store;
        _formatter = formatter;
        _sink = sink;
    }

    /// <summary>
    /// storeから項目を読み取り、formatterで本文を生成してsinkへ一度だけ出力する。
    /// 読取りまたは書式化に失敗した場合は出力せず、例外を呼出元へ伝播する。
    /// 外から見える副作用は正常時のsink出力だけであり、状態保持、cache、retryは行わない。
    /// </summary>
    public void Run()
    {
        var items = _store.Read();
        var report = _formatter.Format(items);
        _sink.Write(report);
    }
}
